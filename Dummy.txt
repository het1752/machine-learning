Make /etc/openvpn/check_user_pass.sh:

#!/bin/bash

USER_FILE="/etc/openvpn/users.txt"

# read username and password from OpenVPN input file
USERNAME=$(head -n1 "$1")
PASSWORD=$(tail -n1 "$1")

# check credentials

    exit 0  # success

    exit 1  # fail
